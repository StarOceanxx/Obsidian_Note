``` bash
# 查看当前网络接口名称
nmcli device status

# 假设你的虚拟机网段是 192.168.1.x，网关是 192.168.1.1
sudo nmcli connection add type ethernet con-name ens7 ifname ens7 \
ipv4.method manual \
ipv4.addresses 192.168.186.37/24 \
ipv4.gateway 192.168.186.1 \
connection.autoconnect yes

# 配置静态IP地址、网关、DNS
sudo nmcli connection modify ens7 ipv4.addresses 192.168.1.100/24  
sudo nmcli connection modify ens7 ipv4.gateway 192.168.1.1  
sudo nmcli connection modify ens7 ipv4.dns "8.8.8.8,114.114.114.114"  
sudo nmcli connection modify ens7 ipv4.method manual

# 激活网卡
sudo nmcli connection up ens7

# 验证IP地址
ip addr show ens7

# 验证DNS配置
cat /etc/resolv.conf

# 测试域名解析
ping -c 3 download.postgresql.org

```
